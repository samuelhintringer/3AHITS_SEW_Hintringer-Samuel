```csharp
namespace OOP;

class Rectangle
{
    double a;
    double b;

    public string ToString()
    {
        return $"a: {a}\nb: {b}";
    }

    public void Resize()
    {
        if(a < b){
            b -= a;
        }
        else a -= b;
    }

    public void Inflate(int p)
    {
        if(p > 100 || p < 0)
        {
            if(p > 100)
            {
                p = 100;
            }
            if(p < 0)
            {
                p = 0;
            }
        }
        a += (a * p / 100);
        b += (b * p / 100);

    }

    public double AspectRatio()
    {
        return a / b;
    }

    public void SetMaxSide(double l)
    {
        if(a > b)
        {
            b = l * AspectRatio();
            a = l;
        }
        else
        {
            a = l * AspectRatio();
            b = l;
        }
    }

    public double Area()
    {
        return a * b;
    }

    public int Floor(double x)
    {
        return (int)x;
    }

    public int Tile(Rectangle r)
    {
        double x = Area();
        return Floor(x /= r.Area());
    }

    public Rectangle() : this(10, 20)
    {

    }

    public Rectangle(double a, double b)
    {
        this.a = a;
        this.b = b;
    }
}

class Program
{
    private static void Main(string[] args)
    {
        Rectangle R = new Rectangle(100, 64);
        Rectangle S = new Rectangle(15.15, 15.15);
        //square.Inflate(50);
        Console.WriteLine(R.Tile(S));
        Console.WriteLine(R.ToString());
    }
}
```

```csharp
class Bankaccount
{
    double balance;
    int interestRate;
    string owner;

    public string ToString()
    {
        return $"Balance: {balance}\nInterest Rate: {interestRate}\nOwner: {owner}";
    }

    public void In(double amount)
    {
        balance += amount;
    }

    public void Out(double amount)
    {
        balance -= amount;
    }

    public double Jahresabschluss()
    {
        balance += balance * interestRate / 100;
        return balance * interestRate / 100;
    }

    public void InterestUp(int num)
    {
        interestRate += (double)num / 10;
    }

    public void InterestDown(int num)
    {
        interestRate -= (double)num / 10;
    }

    public Bankaccount() : this(500, 0.5, "")
    {

    }

    public Bankaccount(double balance, int interestRate, string owner)
    {
        this.balance = balance;
        this.interestRate = interestRate;
        this.owner = owner;
    }
}
```

## 2.4 Bankkonto Erweiterungen

```csharp
class Bankaccount
{
    double balance;
    double interestRate;
    string owner;
    int transCount;

    public string ToString()
    {
        return $"Balance: {balance}\nInterest Rate: {interestRate}\nOwner: {owner}";
    }

    public void In(double amount)
    {
        balance += amount;
    }

    public void Out(double amount)
    {
        balance -= amount;
    }

    public double Jahresabschluss()
    {
        if(transCount > 5)
        {
            if(transCount > 10)
            {
                balance -= (double)(transCount - 10) * 0.05 + 5;
            }
            else
            {
                balance -= (double)(transCount - 5) * 5;
            }
        }
        transCount = 0;
        balance += balance * interestRate / 100;
        return balance * interestRate / 100;
    }

    public void InterestUp(int num)
    {
        interestRate += (double)num / 10;
    }

    public void InterestDown(int num)
    {
        interestRate -= (double)num / 10;
    }

    public void Transfer(Bankaccount Maxi, double amount)
    {
        balance -= amount;
        Maxi.balance += amount;
        transCount++;
    }

    public Bankaccount() : this(500, 0.5, "")
    {

    }

    public Bankaccount(double balance, double interestRate, string owner)
    {
        this.balance = balance;
        this.interestRate = interestRate;
        this.owner = owner;
    }
}
```
